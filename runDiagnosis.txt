async function runDiagnosis() {
  const industry = document.getElementById("industryInput")?.value || "";
  const problem = document.getElementById("problem")?.value || "";
  const outputBox = document.getElementById("output");

  outputBox.textContent = "Processing...";

  try {
    const res = await fetch("/api/diagnose", {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({ industry, problem })
    });

    const text = await res.text();

    // ✅ CRITICAL FIX
    if (!text || text.trim() === "") {
      outputBox.textContent = "Backend returned empty response.";
      return;
    }

    let data;
    try {
      data = JSON.parse(text);
    } catch (e) {
      outputBox.textContent = "Invalid JSON from backend:\n\n" + text;
      return;
    }

    // ✅ SHOW BACKEND ERROR CLEANLY
    if (!res.ok || data.error) {
      outputBox.textContent = "Backend Error:\n\n" + (data.error || text);
      return;
    }

    const safe = (v) => v ?? "";

    let formatted =
      "Executive Summary:\n" + safe(data.executiveSummary) + "\n\n" +
      "Core Problem:\n" + safe(data.coreProblem) + "\n\n" +
      "Root Causes:\n" + (Array.isArray(data.rootCauses) ? data.rootCauses.join("\n") : "") + "\n\n" +
      "System Breakdown:\n" +
        "People:\n" + (Array.isArray(data.systemBreakdown?.people) ? data.systemBreakdown.people.join("\n") : "") + "\n\n" +
        "Process:\n" + (Array.isArray(data.systemBreakdown?.process) ? data.systemBreakdown.process.join("\n") : "") + "\n\n" +
        "Systems:\n" + (Array.isArray(data.systemBreakdown?.systems) ? data.systemBreakdown.systems.join("\n") : "") + "\n\n" +
      "FLOW Framework:\n" +
        "Find:\n" + safe(data.flowFramework?.find) + "\n\n" +
        "Layout:\n" + safe(data.flowFramework?.layout) + "\n\n" +
        "Optimize:\n" + safe(data.flowFramework?.optimize) + "\n\n" +
        "Work:\n" + safe(data.flowFramework?.work) + "\n\n" +
      "Priority / Recommended Actions:\n" + (Array.isArray(data.priorityActions) ? data.priorityActions.join("\n") : "") + "\n\n" +
      "SOP Suggestions:\n" + (Array.isArray(data.sopSuggestions) ? data.sopSuggestions.join("\n") : "") + "\n\n" +
      "KPI Suggestions:\n" + (Array.isArray(data.kpiSuggestions) ? data.kpiSuggestions.join("\n") : "") + "\n\n" +
      "Expected Outcome:\n" + safe(data.expectedOutcome) + "\n\n" +
      "COO Verdict:\n" + safe(data.cooVerdict);

    outputBox.textContent = formatted;

  } catch (err) {
    outputBox.textContent = "Frontend error:\n" + err.message;
  }
}
