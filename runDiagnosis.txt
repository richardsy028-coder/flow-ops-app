async function runDiagnosis(){
  const problem = document.getElementById("problem").value.trim();
  const industry = document.getElementById("industry").value;
  const output = document.getElementById("output");

  if(problem.length < 20){
    alert("Please describe the operational issue in more detail.");
    return;
  }

  output.innerHTML = `
    <div class="output-card">
      <p class="loading">FLOW is analyzing your operations...</p>
    </div>
  `;

  try{
    const response = await fetch("/api/diagnose", {
      method:"POST",
      headers:{ "Content-Type":"application/json" },
      body:JSON.stringify({ problem, industry })
    });

    // ✅ SAFE PARSING (NO MORE CRASH)
    const text = await response.text();

    let data;
    try{
      data = JSON.parse(text);
    }catch(e){
      console.error("Raw response:", text);
      throw new Error(text || "Invalid JSON from backend");
    }

    if(!response.ok || data.error){
      throw new Error(data.error || text || "Backend error");
    }

    // ✅ SAFE DATA HANDLING
    const safeArray = (v) => Array.isArray(v) ? v : [];
    const safeObj = (v) => v && typeof v === "object" ? v : {};

    const renderList = (arr) => {
      const items = safeArray(arr);
      if(!items.length) return "<li>No data returned.</li>";
      return items.map(i => `<li>${typeof i === "string" ? i : (i.issue || i.cause || i.action || i.sop || i.kpi || i.details || JSON.stringify(i))}</li>`).join("");
    };

    const system = safeObj(data.systemBreakdown);
    const flow = safeObj(data.flowFramework);

    // ✅ OUTPUT RENDER
    output.innerHTML = `
      <div class="output-card"><h4>Executive Summary</h4><p>${data.executiveSummary || "No summary returned."}</p></div>
      <div class="output-card"><h4>Core Problem</h4><p>${data.coreProblem || "No core problem returned."}</p></div>

      <div class="output-card"><h4>Root Causes</h4><ul>${renderList(data.rootCauses)}</ul></div>

      <div class="output-card">
        <h4>System Breakdown</h4>
        <p><strong>People:</strong> ${safeArray(system.people).join(", ") || "—"}</p>
        <p><strong>Process:</strong> ${safeArray(system.process).join(", ") || "—"}</p>
        <p><strong>Systems:</strong> ${safeArray(system.systems).join(", ") || "—"}</p>
      </div>

      <div class="output-card">
        <h4>FLOW Framework</h4>
        <p><strong>Find:</strong> ${flow.find || "—"}</p>
        <p><strong>Layout:</strong> ${flow.layout || "—"}</p>
        <p><strong>Optimize:</strong> ${flow.optimize || "—"}</p>
        <p><strong>Work:</strong> ${flow.work || "—"}</p>
      </div>

      <div class="output-card"><h4>Priority / Recommended Actions</h4><ul>${renderList(data.priorityActions || data.recommendedActions)}</ul></div>
      <div class="output-card"><h4>SOP Suggestions</h4><ul>${renderList(data.sopSuggestions)}</ul></div>
      <div class="output-card"><h4>KPI Suggestions</h4><ul>${renderList(data.kpis || data.kpiSuggestions)}</ul></div>
      <div class="output-card"><h4>Expected Outcome</h4><p>${data.expectedOutcome || "No outcome returned."}</p></div>
      <div class="output-card"><h4>COO Verdict</h4><p>${data.cooVerdict || "No verdict returned."}</p></div>
    `;

  }catch(error){
    console.error("REAL ERROR:", error);

    output.innerHTML = `
      <div class="output-card">
        <h4>Backend Error</h4>
        <p>${error.message}</p>
      </div>
    `;
  }
}
