export async function onRequestPost(context) {
  try {
    let body;

    try {
      body = await context.request.json();
    } catch {
      return Response.json(
        { error: "Invalid or empty request body." },
        { status: 400 }
      );
    }

    const industry = body.industry || "General Business";
    const problem = body.problem || "";

    if (!problem.trim()) {
      return Response.json(
        { error: "Please describe the operational problem." },
        { status: 400 }
      );
    }

    return Response.json({
      executiveSummary:
        `This ${industry} operation is experiencing execution friction because the workflow lacks clear ownership, follow-up, and visibility.`,

      coreProblem:
        "The business is relying too much on memory, manual reminders, and inconsistent execution instead of a simple operating system.",

      rootCauses: [
        "Responsibilities are unclear.",
        "The workflow is not documented.",
        "Follow-up depends too much on one person.",
        "There is no simple KPI rhythm to track progress."
      ],

      systemBreakdown: {
        people: [
          "Team members may not know exactly who owns each step.",
          "Accountability is reactive instead of built into the workflow."
        ],
        process: [
          "The current process has unclear handoffs.",
          "Repeated problems are being solved manually instead of systemized."
        ],
        systems: [
          "There is no central tracker or dashboard.",
          "KPIs are missing, ignored, or not reviewed consistently."
        ]
      },

      flowFramework: {
        find: "Find the single biggest bottleneck causing delay, confusion, or rework.",
        layout: "Map the real workflow from start to finish, including who owns each step.",
        optimize: "Remove unnecessary steps, clarify approvals, and simplify communication.",
        work: "Install SOPs, task ownership, weekly review cadence, and KPI tracking."
      },

      priorityActions: [
        "Map the current workflow in 7 steps or fewer.",
        "Assign one clear owner to every recurring task.",
        "Create a weekly operations review.",
        "Track only 1-3 KPIs at first."
      ],

      sopSuggestions: [
        "Task Handoff SOP",
        "Weekly Operations Review SOP",
        "Follow-Up and Escalation SOP"
      ],

      kpiSuggestions: [
        "On-time completion rate",
        "Average turnaround time",
        "Follow-up completion rate"
      ],

      expectedOutcome:
        "The team should gain clearer execution, fewer repeated mistakes, faster follow-up, and better visibility into what is actually happening.",

      cooVerdict:
        "The problem is not effort. The system is leaking. Fix ownership, workflow, and follow-up before blaming people."
    });

  } catch (err) {
    return Response.json(
      { error: err.message || "Server error." },
      { status: 500 }
    );
  }
}
